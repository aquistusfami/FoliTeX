cmake_minimum_required(VERSION 3.21)
project(mem CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
find_package(PkgConfig REQUIRED)
pkg_check_modules(POPPLER REQUIRED IMPORTED_TARGET poppler-qt6)
add_executable(mem harness.cpp)
target_include_directories(mem PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../src)
target_link_libraries(mem PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../build/libfolitex_lib.a Qt6::Widgets Qt6::Test PkgConfig::POPPLER)
