cmake_minimum_required(VERSION 3.24)
project(pb CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_package(PkgConfig REQUIRED)
pkg_check_modules(P REQUIRED IMPORTED_TARGET poppler-qt6)
foreach(v old new)
  add_executable(pb_${v} main.cpp ${SRC_${v}}/pdfview.cpp ${SRC_${v}}/editor.cpp ${SRC_${v}}/editor.h ${SRC_${v}}/pdfview.h)
  target_include_directories(pb_${v} PRIVATE ${SRC_${v}})
  target_link_libraries(pb_${v} Qt6::Widgets PkgConfig::P)
endforeach()
