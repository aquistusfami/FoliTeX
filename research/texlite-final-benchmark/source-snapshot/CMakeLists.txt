cmake_minimum_required(VERSION 3.24)
project(TeXLite VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
find_package(PkgConfig REQUIRED)
pkg_check_modules(POPPLER REQUIRED IMPORTED_TARGET poppler-qt6)
add_library(texlite_lib src/editor.cpp src/pdfview.cpp src/window.cpp)
target_include_directories(texlite_lib PUBLIC src)
target_link_libraries(texlite_lib PUBLIC Qt6::Widgets PkgConfig::POPPLER)
target_compile_options(texlite_lib PRIVATE -Wall -Wextra -Wpedantic)
add_executable(texlite src/main.cpp)
target_link_libraries(texlite PRIVATE texlite_lib)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(texlite_tests tests/tests.cpp)
    target_link_libraries(texlite_tests PRIVATE texlite_lib Qt6::Test)
    add_test(NAME texlite_tests COMMAND texlite_tests)
    set_tests_properties(texlite_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS texlite RUNTIME DESTINATION bin)
