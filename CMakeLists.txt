cmake_minimum_required(VERSION 3.24)
project(FoliTeX VERSION 0.8.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
find_package(PkgConfig REQUIRED)
pkg_check_modules(POPPLER REQUIRED IMPORTED_TARGET poppler-qt6)
add_library(folitex_lib src/palette.cpp src/editor.cpp src/completiondata.cpp src/pdfview.cpp src/window.cpp)
target_include_directories(folitex_lib PUBLIC src)
target_link_libraries(folitex_lib PUBLIC Qt6::Widgets PkgConfig::POPPLER)
target_compile_options(folitex_lib PRIVATE -Wall -Wextra -Wpedantic)
add_executable(folitex src/main.cpp src/resources.qrc)
target_link_libraries(folitex PRIVATE folitex_lib)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(folitex_tests tests/tests.cpp)
    target_link_libraries(folitex_tests PRIVATE folitex_lib Qt6::Test)
    add_test(NAME folitex_tests COMMAND folitex_tests)
    set_tests_properties(folitex_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS folitex RUNTIME DESTINATION bin)
install(FILES packaging/folitex.desktop DESTINATION share/applications)
install(FILES src/folitex.svg DESTINATION share/icons/hicolor/scalable/apps)
